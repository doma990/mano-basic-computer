Original control timings and equations; personal identifiers omitted.

LDA, STA, ISZ, BUN, ADD, DIV, XOR

T0 : AR <- PC
T1 : IR <- M[AR], PC <- PC + 1
T2 : D0..D15 <- Decode IR(12-15), AR <- IR(4-11)

XOR
D0 T3 : DR <- M[AR]
D0 T4 : AC <- DR XOR AC, SC <- 0

ADD
D1 T3 : DR <- M[AR]
D1 T4 : AC <- AC + DR, E <- Cout, SC <- 0

LDA
D2 T3 : DR <- M[AR]
D2 T4 : AC <- DR, SC <- 0

STA
D3 T3 : M[AR] <- AC, SC <- 0

BUN
D4 T3 : PC <- AR, SC <- 0

DIV
D5 T3 : DR <- M[AR]
D5 T4 : TR <- AC
D5 T5 : AC <- DR
D5 T6 : DR <- TR
D5 T7 : AC <- AC / DR, SC <- 0

ISZ
D6 T3 : DR <- M[AR]
D6 T4 : DR <- DR + 1
D6 T5 : M[AR] <- DR, if (DR = 0) then (PC <- PC + 1), SC <- 0
